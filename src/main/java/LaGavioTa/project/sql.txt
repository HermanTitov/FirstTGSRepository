CREATE Table categories(
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name varchar(100) NOT NULL UNIQUE
);

CREATE TABLE ingredients(
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE dishes(
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title VARCHAR(50) UNIQUE NOT NULL,
    description VARCHAR(255) NOT NULL,
    prep_time INTERVAL NOT NULL,
    portion_size INT NOT NULL,
    cost DECIMAL NOT NULL,
    category_id INT REFERENCES categories(id) on DELETE RESTRICT
);

CREATE TABLE dishes_ingredients(
    dish_id int not null references dishes(id) ON DELETE RESTRICT ,
    ingredient_id int not null references ingredients(id) ON DELETE RESTRICT ,
    PRIMARY KEY (dish_id,ingredient_id)
);

drop table dishes cascade ;
drop table dishes_ingredients cascade;
drop table ingredients cascade;
drop table categories cascade;
